function extractNames(){
  const t=names_paste.value; const out=[];
  t.split(/\s+/).forEach(tok=>{
    if(tok.startsWith('magnet:?')){
      const m = /[?&]dn=([^&]+)/.exec(tok); if(m){ out.push(decodeURIComponent(m[1]).replace(/[+]/g,' ')); }
    }else if(/\.(mkv|mp4|avi|flac|m4b|epub|pdf|mp3|aac|zip|rar)$/i.test(tok)){
      out.push(tok.split('/').pop());
    }
  });
  window.EXC = (window.EXC||[]).concat(out);
  renderLists(); UI.toast('Extracted '+out.length+' names');
}
document.addEventListener('DOMContentLoaded', ()=>{ window.INC=[]; window.EXC=[]; renderLists(); });

function toggleRDLists(){
  const d = document.getElementById('rd_side_lists');
  if(!d) return;
  d.classList.toggle('collapsed');
}
document.addEventListener('DOMContentLoaded', ()=>{
  const tb = document.getElementById('pm_rd') || document.body;
  const b = document.createElement('button'); b.className='iconbtn'; b.textContent='Lists';
  b.onclick = toggleRDLists; tb.appendChild(b);
  // Context actions on candidates table: add to include/exclude
  const tbl = document.getElementById('rd_candidates') || document.getElementById('lg_table');
  if(tbl){
    tbl.addEventListener('contextmenu',(e)=>{
      e.preventDefault();
      const row = e.target.closest('tr'); if(!row) return;
      const name = row.dataset.name || row.textContent.trim();
      const m = document.createElement('div'); m.className='ctx'; m.style.left=e.clientX+'px'; m.style.top=e.clientY+'px';
      m.innerHTML = "
Add to Include
Add to Exclude
";
      document.body.appendChild(m); m.style.display='block';
      const done=()=>{ m.remove(); };
      m.addEventListener('click',(ev)=>{
        if(ev.target.id==='ctx_inc'){ window.INC=(window.INC||[]).concat([name]); }
        if(ev.target.id==='ctx_exc'){ window.EXC=(window.EXC||[]).concat([name]); }
        if(typeof renderLists==='function') renderLists(); done();
      });
      document.addEventListener('click', done, {once:true});
    });
  }
});

  window.EXC = (window.EXC||[]).concat(out);
  renderLists(); UI.toast('Extracted '+out.length+' names');
}
document.addEventListener('DOMContentLoaded', ()=>{ window.INC=[]; window.EXC=[]; renderLists(); });

  if(tbl){
    tbl.addEventListener('contextmenu',(e)=>{
      e.preventDefault();
      const row = e.target.closest('tr'); if(!row) return;
      const name = row.dataset.name || row.textContent.trim();
      const m = document.createElement('div'); m.className='ctx'; m.style.left=e.clientX+'px'; m.style.top=e.clientY+'px';
      m.innerHTML = "
Add to Include
Add to Exclude
";
      document.body.appendChild(m); m.style.display='block';
      const done=()=>{ m.remove(); };
      m.addEventListener('click',(ev)=>{
        if(ev.target.id==='ctx_inc'){ window.INC=(window.INC||[]).concat([name]); }
        if(ev.target.id==='ctx_exc'){ window.EXC=(window.EXC||[]).concat([name]); }
        if(typeof renderLists==='function') renderLists(); done();
      });
      document.addEventListener('click', done, {once:true});
    });
  }
});
